Embedded NMPC for Two-Wheeled Inverted Pendulum

Derivation, acados Real-Time Iteration (RTI), and MuJoCo Simulation

1. Mathematical Formulation & System Dynamics

The robot is modeled as an inverted pendulum mounted on two independent coaxial wheels of radius $R$. The forward displacement is denoted by $x$, the torso pitch deviation from vertical by $\theta$, and the motor torque reaction by $\tau$.

Figure 1: Coordinate frames, kinematics, and torque interactions.

Euler-Lagrange Equations of Motion

The generalized coordinates are $q = [x, \theta]^T$. The non-linear dynamics are expressed in standard manipulator form:

$$M(q)\ddot{q} + C(q, \dot{q})\dot{q} + G(q) = B u$$

Where:

$$M(q) = \begin{bmatrix} M + m + \frac{I_w}{R^2} & M l \cos\theta \\ M l \cos\theta & I_y + M l^2 \end{bmatrix}, \quad C(q, \dot{q}) = \begin{bmatrix} 0 & -M l \dot{\theta} \sin\theta \\ 0 & 0 \end{bmatrix}$$ $$G(q) = \begin{bmatrix} 0 \\ -M g l \sin\theta \end{bmatrix}, \quad B = \begin{bmatrix} \frac{1}{R} \\ -1 \end{bmatrix}, \quad u = \tau$$

Defining the state vector $x = [p, v, \theta, \omega]^T$, the explicit first-order non-linear ODE $\dot{x} = f_{\text{expl}}(x, u)$ is obtained via matrix inversion:

$$\begin{bmatrix} \dot{v} \\ \dot{\omega} \end{bmatrix} = M(q)^{-1} \left( B u - C(q, \dot{q})\dot{q} - G(q) \right)$$

2. acados Optimal Control Formulation

We discretize the continuous system over horizon $T_f = 0.5\,\text{s}$ into $N = 25$ steps ($\Delta t = 20\,\text{ms}$, running at $50\,\text{Hz}$). The multiple-shooting NLP minimizes tracking error under strict torque bounds:

$$\min_{x_{0:N}, u_{0:N-1}} \sum_{k=0}^{N-1} \frac{1}{2} \Vert{} y_k - y_{\text{ref}} \Vert{}_{W}^2 + \frac{1}{2} \Vert{} y_N - y_{\text{ref},e} \Vert{}_{W_e}^2$$ $$\text{subject to:} \quad x_0 = \hat{x}_t, \quad x_{k+1} = \Phi_{\text{RK4}}(x_k, u_k), \quad -\tau_{\max} \le u_k \le \tau_{\max}$$
Figure 2: Receding horizon execution principle. Only $u_0^*$ is sent to actuators.
Real-Time Iteration (RTI) Scheme: Rather than iterating an interior-point solver to numerical convergence at each timestep, acados employs an SQP-RTI scheme. The nonlinear problem is linearized during the preparation phase, and a single QP is solved via HPIPM during the feedback phase, yielding solve times under 250 microseconds.

3. Python Code Generation Script (`export_acados.py`)

import os
import casadi as cs
import numpy as np
from acados_template import AcadosModel, AcadosOcp, AcadosOcpSolver

def export_robot_model() -> AcadosModel:
    model_name = "two_wheeled_balancer"
    
    # Dynamics parameters matching robot.xml
    M, m, L, l = 2.0, 0.4, 0.3, 0.15
    I_y = (1.0 / 12.0) * M * (L**2)
    R = 0.05
    I_w = 0.5 * m * (R**2)
    g = 9.81
    
    p = cs.SX.sym('p')
    v = cs.SX.sym('v')
    theta = cs.SX.sym('theta')
    omega = cs.SX.sym('omega')
    x = cs.vertcat(p, v, theta, omega)
    
    tau = cs.SX.sym('tau')
    u = cs.vertcat(tau)
    
    M_tot = M + m + (I_w / (R**2))
    det_M = M_tot * (I_y + M * (l**2)) - (M * l * cs.cos(theta))**2
    
    f_v = (tau / R) + M * l * (omega**2) * cs.sin(theta)
    f_th = M * g * l * cs.sin(theta) - tau
    
    v_dot = ((I_y + M * (l**2)) * f_v - (M * l * cs.cos(theta)) * f_th) / det_M
    omega_dot = (-(M * l * cs.cos(theta)) * f_v + M_tot * f_th) / det_M
    
    f_expl = cs.vertcat(v, v_dot, omega, omega_dot)
    
    model = AcadosModel()
    model.name = model_name
    model.x = x
    model.u = u
    model.f_expl_expr = f_expl
    model.xdot = cs.SX.sym('xdot', 4)
    model.f_impl_expr = model.xdot - f_expl
    return model

def generate_solver():
    ocp = AcadosOcp()
    model = export_robot_model()
    ocp.model = model
    
    Tf, N = 0.5, 25
    ocp.dims.N = N
    ocp.solver_options.tf = Tf
    
    nx, nu = 4, 1
    ny, ny_e = nx + nu, nx
    ocp.cost.cost_type = 'LINEAR_LS'
    ocp.cost.cost_type_e = 'LINEAR_LS'
    
    Q = np.array([10.0, 1.0, 50.0, 5.0])
    R_weight = np.array([0.05])
    ocp.cost.W = np.diag(np.concatenate([Q, R_weight]))
    ocp.cost.W_e = np.diag(Q * 2.0)
    
    ocp.cost.Vx = np.zeros((ny, nx))
    ocp.cost.Vx[:nx, :nx] = np.eye(nx)
    ocp.cost.Vu = np.zeros((ny, nu))
    ocp.cost.Vu[nx:, :] = np.eye(nu)
    ocp.cost.Vx_e = np.eye(ny_e)
    
    ocp.cost.yref = np.zeros(ny)
    ocp.cost.yref_e = np.zeros(ny_e)
    
    ocp.constraints.lbu = np.array([-1.5])
    ocp.constraints.ubu = np.array([1.5])
    ocp.constraints.idxbu = np.array([0])
    ocp.constraints.x0 = np.zeros(nx)
    
    ocp.solver_options.qp_solver = 'PARTIAL_CONDENSING_HPIPM'
    ocp.solver_options.hessian_approx = 'GAUSS_NEWTON'
    ocp.solver_options.integrator_type = 'ERK'
    ocp.solver_options.nlp_solver_type = 'SQP_RTI'
    
    solver = AcadosOcpSolver(ocp, json_file='acados_ocp_balancer.json')
    print("Code generation complete in c_generated_code/")

if __name__ == '__main__':
    generate_solver()

4. C++ Execution in MuJoCo Harness

#include <mujoco/mujoco.h>
#include <iostream>
#include <chrono>
#include "acados_solver_two_wheeled_balancer.h"

int main() {
    mjModel* m = mj_loadXML("robot.xml", nullptr, nullptr, 0);
    mjData* d = mj_makeData(m);

    // Initialize acados preallocated C capsule
    auto* capsule = two_wheeled_balancer_acados_create_capsule();
    two_wheeled_balancer_acados_create(capsule);
    
    auto* config = two_wheeled_balancer_acados_get_nlp_config(capsule);
    auto* dims = two_wheeled_balancer_acados_get_nlp_dims(capsule);
    auto* in = two_wheeled_balancer_acados_get_nlp_in(capsule);
    auto* out = two_wheeled_balancer_acados_get_nlp_out(capsule);

    const int decimation = 10; // 500 Hz physics / 10 = 50 Hz MPC
    int steps = 0;
    double u_opt = 0.0;

    while (d->time < 10.0) {
        if (steps % decimation == 0) {
            // Extract state: [x, v, theta, omega]
            double pos = d->qpos[0];
            double vel = d->qvel[0];
            double pitch = 2.0 * std::asin(d->qpos[5]);
            double pitch_rate = d->qvel[4];
            double x0[4] = {pos, vel, pitch, pitch_rate};

            ocp_nlp_constraints_model_set(config, dims, in, 0, "lbx", x0);
            ocp_nlp_constraints_model_set(config, dims, in, 0, "ubx", x0);

            // Solve single RTI iteration
            two_wheeled_balancer_acados_solve(capsule);
            ocp_nlp_out_get(config, dims, out, 0, "u", &u_opt);
        }

        d->ctrl[0] = 0.5 * u_opt;
        d->ctrl[1] = 0.5 * u_opt;
        mj_step(m, d);
        steps++;
    }

    two_wheeled_balancer_acados_free(capsule);
    two_wheeled_balancer_acados_free_capsule(capsule);
    mj_deleteData(d);
    mj_deleteModel(m);
    return 0;
}
